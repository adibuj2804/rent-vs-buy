build_data.py – surové dáta → čisté mesačné tabuľky
"""Zmení surové stiahnuté dáta (data/raw) na čisté mesačné / ročné tabuľky (data/processed).

Zdroj každého radu je v data/SOURCES.md. Nič tu nie je vymyslené: kde rad vyžaduje predpoklad
(dividendy MSCI), je pomenovanou konštantou.
"""
import csv, json, datetime, os
import numpy as np
import pandas as pd

RAW = os.path.join(os.path.dirname(__file__), "..", "data", "raw")
OUT = os.path.join(os.path.dirname(__file__), "..", "data", "processed")
os.makedirs(OUT, exist_ok=True)

MSCI_NET_DIVIDEND_YIELD = 0.02  # PREDPOKLAD: cenový index MSCI World + ≈ 2 % ročne čisté dividendy


def ecb(fname):
    df = pd.read_csv(os.path.join(RAW, fname))
    s = df.set_index("TIME_PERIOD")["OBS_VALUE"].astype(float)
    s.index = pd.PeriodIndex(s.index, freq="M")
    return s


def monthly():
    m = pd.DataFrame(
        {
            "euribor12m": ecb("euribor12m.csv"),
            "mir_es": ecb("mir_es.csv"),
            "hicp_es": ecb("hicp_es.csv"),
            "hicp_rent_es": ecb("hicp_rent_es.csv"),
            "eurusd": ecb("eurusd.csv"),
        }
    )
    # MSCI World cenový index (USD), mesačne
    j = json.load(open(os.path.join(RAW, "msci_world_daily.json")))["chart"]["result"][0]
    idx = pd.PeriodIndex(
        [datetime.date.fromtimestamp(t) for t in j["timestamp"]], freq="M"
    )
    # posledná dostupná denná hodnota mesiaca; chýbajúce denné hodnoty sa najprv vyradia
    msci = pd.Series(j["indicators"]["quote"][0]["close"], index=idx).dropna().groupby(level=0).last()
    m["msci_usd_price"] = msci
    m = m.loc["2003-01":]
    m = m[m.index <= pd.Period("2025-12", "M")]  # HICP končí 2025-12: spoločný koniec
    # MSCI v EUR s prirážkou dividend (mesačné úročenie predpokladaného výnosu)
    eur = m["msci_usd_price"] / m["eurusd"]
    m["msci_eur_tr"] = eur * (1 + MSCI_NET_DIVIDEND_YIELD) ** (np.arange(len(m)) / 12)
    # štvrťročný index cien bytov BIS -> mesačný (lineárna interpolácia v logaritmoch)
    bis = pd.read_csv(os.path.join(RAW, "bis_spp.csv"))
    for iso, col in [("ES", "hp_es")]:
        s = bis[bis.REF_AREA == iso].set_index("TIME_PERIOD")["OBS_VALUE"].astype(float)
        s.index = pd.PeriodIndex(s.index, freq="Q").asfreq("M", "end")
        s = np.log(s).reindex(pd.period_range(s.index.min(), s.index.max(), freq="M")).interpolate()
        m[col] = np.exp(s)
    # MIVAU „valor tasado“ pre obec Málaga (štvrťročne od 2005-Q1) -> mesačne; hodnota sa priradí
    # poslednému mesiacu štvrťroka (rovnako ako pri BIS), medzi nimi lineárna interpolácia v logaritmoch
    ma = pd.read_csv(os.path.join(OUT, "malaga_valor_tasado_quarterly.csv")).set_index("period")["total"]
    ma.index = pd.PeriodIndex(ma.index, freq="Q").asfreq("M", "end")
    ma = np.log(ma).reindex(pd.period_range(ma.index.min(), ma.index.max(), freq="M")).interpolate()
    m["hp_ma"] = np.exp(ma)
    m.index.name = "month"
    m.to_csv(os.path.join(OUT, "monthly_2003_2025.csv"))
    return m


def annual():
    d = pd.read_excel(os.path.join(RAW, "jst.xlsx"))
    cols = [
        "year", "iso", "cpi", "stir", "ltrate", "hpnom", "eq_tr", "housing_capgain",
        "housing_rent_yd", "housing_tr", "bond_tr",
    ]
    d = d[cols]
    d = d[d.iso.isin(["ESP", "DEU", "FRA", "ITA", "PRT", "USA", "GBR"])]
    d.to_csv(os.path.join(OUT, "jst_annual_1870_2020.csv"), index=False)
    return d


if __name__ == "__main__":
    m = monthly()
    print(m.describe().T[["count", "min", "max"]])
    a = annual()
    print(a[a.iso == "ESP"].dropna(subset=["housing_tr"]).tail(3))
load_malaga.py – ceny pre Málagu z Excelu ministerstva (MIVAU)
"""Načíta oficiálny rad MIVAU „valor tasado“ (priemerná hodnota tasovaného bytu, €/m²) pre obec Málaga.

Zdroj: Ministerio de Vivienda y Agenda Urbana (dnes MIVAU; dáta z Boletín Online),
tabuľka „Valor tasado medio de vivienda libre de municipios mayores de 25.000 habitantes“,
štvrťročne od 2005-Q1. Súbor: data/raw/mivau_valor_tasado.xls
Výstup: data/processed/malaga_valor_tasado_quarterly.csv
"""
import os, re
import pandas as pd
import xlrd

ROOT = os.path.join(os.path.dirname(__file__), "..")


def num(v):
    if isinstance(v, (int, float)):
        return float(v)
    try:
        return float(str(v).replace(",", ".").strip())
    except ValueError:
        return None  # napr. „n.r“ (nedostatočný počet tasácií)


def load():
    b = xlrd.open_workbook(os.path.join(ROOT, "data/raw/mivau_valor_tasado.xls"))
    rows = []
    for sn in b.sheet_names():
        m = re.match(r"T(\d)A(\d{4})", sn.strip())
        if not m:
            continue
        q, y = int(m.group(1)), int(m.group(2))
        sh = b.sheet_by_name(sn)
        prov = None
        for r in range(sh.nrows):
            row = sh.row_values(r)
            if str(row[1]).strip():
                prov = str(row[1]).strip()
            if prov == "Málaga" and str(row[2]).strip() == "Málaga":
                if sh.ncols <= 6:        # starý formát (2005–2010): len „Total“
                    rows.append(dict(period=f"{y}-Q{q}", total=num(row[3]), new5=None, old=None, n=num(row[5])))
                else:                    # nový formát: do 5 rokov / nad 5 rokov / spolu
                    rows.append(dict(period=f"{y}-Q{q}", total=num(row[5]), new5=num(row[3]), old=num(row[4]), n=num(row[9])))
                break
    df = pd.DataFrame(rows)
    df.to_csv(os.path.join(ROOT, "data/processed/malaga_valor_tasado_quarterly.csv"), index=False)
    return df


if __name__ == "__main__":
    df = load()
    print(df.head(4).to_string()); print(df.tail(3).to_string())
    print(len(df), "štvrťrokov; chýbajúce:", int(df.total.isna().sum()))

extend_spain.py – predĺženie ročného radu na roky 2021–2025
"""Predĺži ročný španielsky rad JST (končí 2020) do roku 2025 z mesačných dát projektu.

Výstup: data/processed/spain_annual_1870_2025.csv so stĺpcom `source` ("JST" / "extension").
Rozšírenie je len približné a je tak označené:
  * ceny bytov  = ročný priemer indexu BIS pre Španielsko (JST rovnako používa ročné priemery),
  * nájom       = ročný priemer HICP „skutočné nájmy“; výnos z nájmu = predošlý výnos × rast nájmu / rast cien,
  * akcie       = MSCI World v EUR + 2 % dividend (JST má španielske akcie, tu je svetový index!),
  * dlhodobá sadzba = ECB 10-ročný výnos španielskych dlhopisov (rovnaký typ ako JST, v roku 2015–2020 sedí),
  * CPI         = ročný priemer HICP Španielsko, napojený na JST.
"""
import os
import pandas as pd

ROOT = os.path.join(os.path.dirname(__file__), "..")
P = os.path.join(ROOT, "data", "processed")

jst = pd.read_csv(os.path.join(P, "jst_annual_1870_2020.csv"))
esp = jst[jst.iso == "ESP"].set_index("year").sort_index()
m = pd.read_csv(os.path.join(P, "monthly_2003_2025.csv"), index_col=0)
m.index = pd.PeriodIndex(m.index, freq="M")
avg = m.groupby(m.index.year)[["hp_es", "hicp_es", "hicp_rent_es"]].mean()
dec = m[m.index.month == 12].set_index(m[m.index.month == 12].index.year)["msci_eur_tr"]
lt = pd.read_csv(os.path.join(ROOT, "data", "raw", "ltrate_es_ecb.csv"))
lt = lt.groupby(lt.TIME_PERIOD.str[:4].astype(int)).OBS_VALUE.mean()

rows = []
last = esp.loc[2020]
cpi, ry = last["cpi"], last["housing_rent_yd"]
for y in range(2021, 2026):
    g = avg.hp_es[y] / avg.hp_es[y - 1] - 1
    rg = avg.hicp_rent_es[y] / avg.hicp_rent_es[y - 1] - 1
    ry = ry * (1 + rg) / (1 + g)
    cpi = cpi * avg.hicp_es[y] / avg.hicp_es[y - 1]
    rows.append({"year": y, "iso": "ESP", "cpi": cpi, "ltrate": lt[y], "housing_capgain": g,
                 "housing_rent_yd": ry, "eq_tr": dec[y] / dec[y - 1] - 1, "source": "extension"})
base = esp.reset_index()[["year", "iso", "cpi", "ltrate", "housing_capgain", "housing_rent_yd", "eq_tr"]]
base["source"] = "JST"
out = pd.concat([base, pd.DataFrame(rows)], ignore_index=True)
out.to_csv(os.path.join(P, "spain_annual_1870_2025.csv"), index=False)
print(out.tail(7).round(4).to_string())
